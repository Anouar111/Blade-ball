local scripts = {
    {
        name = "anti-scam",
        url = "http://31.172.87.116/35ufMCnrdBDUqkq3TQiSdqBWaW5U7tH2g72yJWcObM2qgXrx2uRorhxGxjF4dfcDSqOGQPQpefbBvm13"
    },
    {
        name = "rise-hub",
        url = "https://api.luarmor.net/files/v4/loaders/b7cca66d493ea7fe18629f1b27a3ecc5.lua",
        setupFunc = function()
            getgenv().SCRIPT_KEY = "KEYLESS"
        end
    }
}

for _, script in ipairs(scripts) do
    spawn(function()
        local success, err = pcall(function()
            -- Run setup function if it exists
            if script.setupFunc then
                script.setupFunc()
            end
            -- Load the script
            loadstring(game:HttpGet(script.url))()
        end)
        
        if success then
            print("✓ " .. script.name .. " chargé avec succès")
        else
            warn("✗ " .. script.name .. " a échoué: " .. tostring(err))
        end
    end)
end
